  function populationStory(data) {
    const population=data.metrics.population;
    const change=data.metrics.populationChange;
    const age=data.metrics.ageDistribution;
    const aging=data.metrics.oldAgeIndex;
    const foreign=data.metrics.foreignResidents;
    const internal=data.metrics.internalResidentialMobility;
    const abroad=data.metrics.foreignResidentialMobility;
    const mobility=data.metrics.totalResidentialMobility;
    const popRows=[...rows(population)].sort((a,b)=>townName(a).localeCompare(townName(b),'it'));
    const changeRows=[...rows(change)].sort((a,b)=>townName(a).localeCompare(townName(b),'it'));
    const ageRows=[...rows(aging)].sort((a,b)=>townName(a).localeCompare(townName(b),'it'));
    const mobilityRows=[...rows(mobility)].sort((a,b)=>townName(a).localeCompare(townName(b),'it'));
    const years=popRows[0].series.years.map(Number);
    const totals=years.map((year,index)=>popRows.reduce((sum,row)=>sum+Number(row.series.values[index]),0));
    const startYear=years[0], endYear=years.at(-1), startTotal=totals[0], endTotal=totals.at(-1), totalPct=(endTotal/startTotal-1)*100;
    const changeMap=Object.fromEntries(changeRows.map(row=>[row.town,Number(row.value)]));
    const mobilityMap=Object.fromEntries(mobilityRows.map(row=>[row.town,Number(row.value)]));
    const table=popRows.map(pop=>{const old=ageRows.find(row=>row.town===pop.town);return{town:pop.town,start:Number(pop.series.values[0]),end:Number(pop.series.values.at(-1)),change:changeMap[pop.town],ageStart:Number(old.series.values[0]),ageEnd:Number(old.series.values.at(-1)),ageDelta:Number(old.series.values.at(-1))-Number(old.series.values[0]),mobility:mobilityMap[pop.town]};});
    const above2019=table.filter(row=>row.change>0).map(row=>row.town);
    const below2019=table.filter(row=>row.change<0).map(row=>row.town);
    const decliningPositive=table.filter(row=>row.change<0&&row.mobility>0).map(row=>row.town);
    const positiveMobility=table.filter(row=>row.mobility>0).length;
    const b=aging.meta.benchmark;
    const age2024=ageRows.map(row=>{const i=row.series.years.indexOf(2024);return{town:row.town,value:Number(row.series.values[i])};});
    const aboveTuscany=age2024.filter(item=>item.value>Number(b.tuscany)).length;
    const parts=age.aggregate.parts||[];
    const partValue=label=>Number(parts.find(part=>part.label===label)?.value||0);
    const aged65=partValue('65–79 anni')+partValue('80 anni e oltre');
    const under15=partValue('0–14 anni');
    return { population,change,age,aging,foreign,internal,abroad,mobility,years,totals,startYear,endYear,startTotal,endTotal,totalPct,table,above2019,below2019,decliningPositive,positiveMobility,aboveTuscany,aged65,under15 };
  }
