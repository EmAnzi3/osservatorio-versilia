      return `${label} è ${diff >= 0 ? 'sopra' : 'sotto'} il riferimento Versilia di ${n1.format(Math.abs(diff))} punti percentuali.`;
    }
    if (['years','minutes','per1000','per100','per10k','per100k','index'].includes(unit)) {
      return `${label} è ${diff >= 0 ? 'sopra' : 'sotto'} il riferimento Versilia di ${formatValue(Math.abs(diff), unit)}.`;
    }
    if (aggregate !== 0) {
      const relative = diff / Math.abs(aggregate) * 100;
      return `${label} è ${relative >= 0 ? 'sopra' : 'sotto'} il riferimento Versilia del ${n1.format(Math.abs(relative))}%.`;
    }
    return `${label}: ${formatValue(local, unit)}; riferimento Versilia ${formatValue(aggregate, unit)}.`;
  }

  function seriesDelta(metric, row) {
    const series = row?.series;
    if (!series?.values?.length || series.values.length < 2) return null;
    const start = number(series.values[0]), end = number(series.values.at(-1));
    if (start === null || end === null) return null;
    const unit = metric.meta.unit;
    const absolute = end - start;
    const relative = start !== 0 ? absolute / Math.abs(start) * 100 : null;
    return { start, end, startYear: series.years[0], endYear: series.years.at(-1), absolute, relative, unit };
  }

  function trendText(metric, row, subject = '') {
    const trend = seriesDelta(metric, row);
    if (!trend) return '';
    const prefix = subject ? `${subject}: ` : '';
    if (trend.unit === 'percent' || trend.unit === 'percentagePoints') {
      return `${prefix}${formatValue(trend.start, trend.unit)} nel ${trend.startYear} e ${formatValue(trend.end, trend.unit)} nel ${trend.endYear}, una variazione di ${sign(trend.absolute)}${n1.format(trend.absolute)} punti.`;
    }
    if (['index','minutes','years','per1000','per100','per10k','per100k'].includes(trend.unit)) {
      return `${prefix}${formatValue(trend.start, trend.unit)} nel ${trend.startYear} e ${formatValue(trend.end, trend.unit)} nel ${trend.endYear}; differenza ${sign(trend.absolute)}${formatValue(trend.absolute, trend.unit)}.`;
    }
    return `${prefix}${formatValue(trend.start, trend.unit)} nel ${trend.startYear} e ${formatValue(trend.end, trend.unit)} nel ${trend.endYear}${trend.relative === null ? '' : `, pari a ${sign(trend.relative)}${n1.format(trend.relative)}%`}.`;
  }

  function benchmarkText(metric, row, town) {
    const benchmark = metric?.meta?.benchmark;
    if (!benchmark || !row) return '';
    const local = number(row.benchmarkValue ?? row.value);
    if (local === null) return '';
    const parts = [];
    if (benchmark.tuscany !== null && benchmark.tuscany !== undefined) {
      const diff = local - Number(benchmark.tuscany);
      parts.push(`${town} ${formatValue(local, metric.meta.unit)}, Toscana ${formatValue(benchmark.tuscany, metric.meta.unit)} (${diff >= 0 ? '+' : '−'}${formatValue(Math.abs(diff), metric.meta.unit)})`);
    }
    if (benchmark.italy !== null && benchmark.italy !== undefined) parts.push(`Italia ${formatValue(benchmark.italy, metric.meta.unit)}`);
    return parts.join('; ') + ` · ${benchmark.year}.`;
  }
